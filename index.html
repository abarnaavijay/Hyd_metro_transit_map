// ================================================================= // ADD THIS SCRIPT TO YOUR qgis2web Leaflet index.html (before ) // This will calculate accurate distances between any tourist spot // and all Hyderabad Metro stations dynamically! // ================================================================= // 1. Gather all metro stations from your existing GeoJSON layers var allStations = []; [json_Red_Stations_10, json_Green_Stations_11, json_Blue_Stations_12].forEach(function(collection) { if (!collection || !collection.features) return; collection.features.forEach(function(feat) { var coords = feat.geometry.coordinates; // Leaflet uses [lat, lng], GeoJSON uses [lng, lat] var lat = feat.geometry.type === 'MultiPoint' ? coords[0][1] : coords[1]; var lng = feat.geometry.type === 'MultiPoint' ? coords[0][0] : coords[0]; allStations.push({ name: feat.properties.Name || 'Metro Station', latLng: L.latLng(lat, lng) }); }); }); // 2. Function to compute accurate distance and walking time function getNearestMetroInfo(targetLatLng) { var nearest = null; var minDistanceMeters = Infinity; allStations.forEach(function(stn) { var d = targetLatLng.distanceTo(stn.latLng); // Built-in Leaflet geodesic distance in meters! if (d < minDistanceMeters) { minDistanceMeters = d; nearest = stn; } }); if (!nearest) return null; var distanceKm = (minDistanceMeters / 1000).toFixed(2); var walkingKm = (distanceKm * 1.25).toFixed(2); // Urban street circuity factor var walkingMins = Math.round((walkingKm / 4.8) * 60); // 4.8 km/h walking speed var drivingMins = Math.max(3, Math.round((distanceKm * 1.3 / 22) * 60) + 2); return { stationName: nearest.name, distanceKm: distanceKm, walkingMins: walkingMins, drivingMins: drivingMins }; } // 3. Update all attraction popups dynamically or draw movement rays on click! map.on('click', function(e) { var info = getNearestMetroInfo(e.latlng); if (!info) return; L.popup() .setLatLng(e.latlng) .setContent( '
' + 'Nearest Metro Movement:
' + '🚉 ' + info.stationName.split(',')[0] + '
' + '📏 Distance: ' + info.distanceKm + ' km
' + '🚶 Walk: ~' + info.walkingMins + ' mins | 🛺 Auto: ~' + info.drivingMins + ' mins' + '
' ) .openOn(map); });
